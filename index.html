from pyodide.ffi import create_proxy def add_numbers(a, b): return a + b def sub_numbers(a, b): return a - b def get_values(): return (float(Element("num1").element.value), float(Element("num2").element.value)) def show(formula, value): Element("formula").write(formula) Element("result").write(value) Element("error").write("") def on_add(e=None): try: a, b = get_values() show("{} + {}".format(a, b), add_numbers(a, b)) except ValueError: Element("error").write("Enter both numbers first") except Exception as ex: Element("error").write("Error: {}".format(ex)) def on_sub(e=None): try: a, b = get_values() show("{} - {}".format(a, b), sub_numbers(a, b)) except ValueError: Element("error").write("Enter both numbers first") except Exception as ex: Element("error").write("Error: {}".format(ex)) Element("btn-add").element.addEventListener("click", create_proxy(on_add)) Element("btn-sub").element.addEventListener("click", create_proxy(on_sub))
